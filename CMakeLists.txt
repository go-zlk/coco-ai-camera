cmake_minimum_required(VERSION 3.18)
project(coco_context VERSION 0.1.0 LANGUAGES CXX)

option(COCO_TENSORRT "Build native TensorRT 10 detector" OFF)
option(COCO_BUILD_RUNTIME "Build camera/inference runtime and its tests" ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(cmake/CompilerOptions.cmake)
include(CTest)
find_package(Threads REQUIRED)
find_package(SQLite3 REQUIRED)

function(coco_module target)
  add_library(${target} STATIC ${ARGN})
  target_compile_features(${target} PUBLIC cxx_std_17)
  target_include_directories(${target} PUBLIC "${PROJECT_SOURCE_DIR}/include")
  target_link_libraries(${target} PRIVATE coco_options)
endfunction()

coco_module(coco_domain src/domain/context.cc src/domain/presence_state.cc src/domain/context_engine.cc src/domain/pet_activity.cc src/domain/utc_time.cc)
add_library(coco::domain ALIAS coco_domain)
coco_module(coco_storage src/storage/event_store.cc)
target_link_libraries(coco_storage PUBLIC coco_domain SQLite::SQLite3 PRIVATE Threads::Threads)
add_library(coco::storage ALIAS coco_storage)
coco_module(coco_transport src/transport/context_api.cc)
target_link_libraries(coco_transport PRIVATE Threads::Threads)
add_library(coco::transport ALIAS coco_transport)

coco_module(coco_tracking src/perception/box_tracker.cc)
target_link_libraries(coco_tracking PUBLIC coco_domain)
add_library(coco::tracking ALIAS coco_tracking)

coco_module(coco_configuration src/application/service_config.cc)
add_library(coco::configuration ALIAS coco_configuration)
coco_module(coco_replay src/application/replay_service.cc src/application/tracking_replay.cc)
target_link_libraries(coco_replay PRIVATE coco_domain coco_storage coco_tracking)
add_library(coco::replay ALIAS coco_replay)

target_link_libraries(coco-replay PRIVATE coco_replay coco_options)

coco_module(coco_dashboard src/application/dashboard.cc)
target_link_libraries(coco_dashboard PUBLIC coco_storage coco_transport)

target_link_libraries(coco-poc PRIVATE coco_poc coco_options)
install(TARGETS coco-poc RUNTIME DESTINATION bin)

if(COCO_BUILD_RUNTIME)
  find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs videoio dnn)
  add_library(coco_media INTERFACE)
  target_include_directories(coco_media INTERFACE "${PROJECT_SOURCE_DIR}/include")
  target_include_directories(coco_media SYSTEM INTERFACE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(coco_media INTERFACE coco_domain opencv_core)
  coco_module(coco_capture src/capture/camera_source.cc)
  target_link_libraries(coco_capture PUBLIC coco_media PRIVATE opencv_videoio Threads::Threads)
  add_library(coco::capture ALIAS coco_capture)
  coco_module(coco_perception src/perception/detector.cc src/perception/pet_appearance.cc)
  target_link_libraries(coco_perception PUBLIC coco_media PRIVATE opencv_imgproc opencv_imgcodecs opencv_dnn)
  add_library(coco::perception ALIAS coco_perception)
  if(COCO_TENSORRT)
    find_package(CUDAToolkit REQUIRED)
    find_path(TENSORRT_INCLUDE NvInfer.h PATH_SUFFIXES aarch64-linux-gnu REQUIRED)
    find_library(TENSORRT_LIB nvinfer REQUIRED)
    target_sources(coco_perception PRIVATE src/perception/tensorrt_detector.cc)
    target_include_directories(coco_perception SYSTEM PRIVATE ${TENSORRT_INCLUDE})
    target_link_libraries(coco_perception PRIVATE ${TENSORRT_LIB} CUDA::cudart)
    target_compile_definitions(coco_perception PRIVATE COCO_TENSORRT=1)
  endif()
  coco_module(coco_application src/application/context_service.cc)
  target_link_libraries(coco_application PUBLIC coco_configuration PRIVATE coco_dashboard coco_capture coco_perception coco_tracking coco_storage coco_transport Threads::Threads)
  add_library(coco::application ALIAS coco_application)
  add_executable(coco-context apps/context_service/main.cc)
  target_link_libraries(coco-context PRIVATE coco_application coco_options)
  install(TARGETS coco-context RUNTIME DESTINATION bin)
