cmake_minimum_required(VERSION 3.18)
project(coco_context VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(COCO_TENSORRT "Build native TensorRT 10 detector" OFF)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc videoio dnn)
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
add_library(coco_core
  cpp/src/types.cpp cpp/src/source.cpp cpp/src/detector.cpp
  cpp/src/state.cpp cpp/src/store.cpp cpp/src/api.cpp)
target_include_directories(coco_core PUBLIC cpp/include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(coco_core PUBLIC ${OpenCV_LIBS} SQLite::SQLite3 Threads::Threads)
target_compile_options(coco_core PRIVATE -Wall -Wextra -Wpedantic)
if(COCO_TENSORRT)
  find_package(CUDAToolkit REQUIRED)
  find_path(TENSORRT_INCLUDE NvInfer.h PATH_SUFFIXES aarch64-linux-gnu REQUIRED)
  find_library(TENSORRT_LIB nvinfer REQUIRED)
  target_sources(coco_core PRIVATE cpp/src/tensorrt_detector.cpp)
  target_include_directories(coco_core PRIVATE ${TENSORRT_INCLUDE})
  target_link_libraries(coco_core PUBLIC ${TENSORRT_LIB} CUDA::cudart)
  target_compile_definitions(coco_core PUBLIC COCO_TENSORRT=1)
endif()
add_executable(coco-context cpp/src/main.cpp)
target_link_libraries(coco-context PRIVATE coco_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(coco-state-test cpp/tests/state_test.cpp)
  target_link_libraries(coco-state-test PRIVATE coco_core)
  add_test(NAME state-and-store COMMAND coco-state-test)
endif()
